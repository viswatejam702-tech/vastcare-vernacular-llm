# Step 3: Quality Filtering Logic
def filter_dataset(dataset):
    filtered_data = []
    for sample in dataset:
        # Check length constraints
        if len(sample['instruction']) < 10 or len(sample['output_te']) < 10:
            continue
            
        # Evaluate translation quality (pseudo-BLEU)
        bleu_score = calculate_bleu(sample['output_en'], sample['output_te'])
        if bleu_score < 0.45:
            log_rejection(sample, reason='Low BLEU')
            continue
            
        filtered_data.append(sample)
        
    return filtered_data
